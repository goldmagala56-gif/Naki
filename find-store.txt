// ---------- sound effects: the sound files behind the effect clips (kept on this device with the session) ----------
S.sfx = {};
function sfxUrl(id){ var o = S.sfx[id]; return o ? o.url : ''; }
function setSfxFile(id, file){
  var old = S.sfx[id]; if (old && old.url) URL.revokeObjectURL(old.url);
  S.sfx[id] = { file: file, url: URL.createObjectURL(file) };
  if (DB.ok && S.sess) DB.put('sfx', { sid: S.sess.id, n: +String(id).replace('fx', ''), id: id, name: file.name, blob: file }).catch(function(){
    toast('This sound is too large to remember on this device. You will need to add it again next time.', 8000);
  });
}
function loadSfx(s){
  Object.keys(S.sfx).forEach(function(k){ try { if (S.sfx[k].url) URL.revokeObjectURL(S.sfx[k].url); } catch (e) {} });
  S.sfx = {};
  if (!DB.ok) return;
  DB.range('sfx', s.id).then(function(list){
    list.forEach(function(r){
      if (!r || !r.blob) return;
      var f = new File([r.blob], r.name || r.id, { type: r.blob.type });
      S.sfx[r.id] = { file: f, url: URL.createObjectURL(f) };
    });
  }, function(){});
}
function pickSfx(){
  return new Promise(function(resolve){
    var inp = $('sfxPick'); inp.value = '';
    inp.onchange = function(){
      var f = inp.files && inp.files[0]; if (!f) return resolve(null);
      var a = new Audio(), u = URL.createObjectURL(f); a.preload = 'metadata';
      a.onloadedmetadata = function(){ URL.revokeObjectURL(u); resolve({ file: f, name: f.name, durMs: isFinite(a.duration) ? Math.round(a.duration * 1000) : 3000 }); };
      a.onerror = function(){ URL.revokeObjectURL(u); toast('This sound file could not be read. Try an MP3, M4A or WAV.'); resolve(null); };
      a.src = u;
    };
    inp.click();
  });
}
